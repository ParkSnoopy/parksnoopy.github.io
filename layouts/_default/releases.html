{{- /* Lists the newest release of each project in front matter `projects`, one block per project, one row per platform. The data comes from the GitHub API at build time. Asset names look like Name_v1.2.3-amd64-Linux.tar.gz or ...-arm64-iOS_Simulator.zip. */ -}} {{ define "main" }} {{- $repo := .Params.repo -}} {{- $headers := dict "Accept" "application/vnd.github+json" -}} {{- with os.Getenv "HUGO_GITHUB_TOKEN" -}} {{- $headers = merge $headers (dict "Authorization" (printf "Bearer %s" .)) -}} {{- end -}} {{- $url := printf "https://api.github.com/repos/%s/releases?per_page=100" $repo -}} {{- $releases := slice -}} {{- with try (resources.GetRemote $url (dict "headers" $headers)) -}} {{- with .Err -}} {{- errorf "Could not read the release list from %s: %s" $url . -}} {{- else with .Value -}} {{- $releases = .Content | transform.Unmarshal -}} {{- else -}} {{- errorf "No release list came back from %s" $url -}} {{- end -}} {{- end -}} {{- $groups := slice (dict "key" "Linux" "label" "Linux") (dict "key" "Windows" "label" "Windows") (dict "key" "Android" "label" "Android") (dict "key" "macOS" "label" "macOS") (dict "key" "" "label" "Others") -}}
